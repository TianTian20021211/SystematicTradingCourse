\section{Setup}

\begin{frame}{Project 7: what we study}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item A futures curve prices the \key{same commodity} at different delivery months
  \item A calendar spread trades two points on that curve
  \[
    C_t = F(t,T_1) - F(t,T_2), \qquad T_1 < T_2
  \]
  \item Long the spread: long the near contract, short the far one; P\&L is
        $q\,M\,\Delta C_{t+h}$
  \item The flat price largely cancels; what remains is the \key{shape of the curve},
        roll and convergence
  \item Research question: does term-structure information predict the spread
        change, \key{net of rolls and costs}?
\end{itemize}
\end{column}

\section{A: Carry timing}

\begin{frame}{A \textperiodcentered\ Carry: the assignment direction, then the flip}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.52\textwidth}
\footnotesize
\textbf{Original A} (assignment Example 1)
\begin{itemize}
  \item Backwardation ($C_t>0$) $\rightarrow$ long spread;
        contango ($C_t<0$) $\rightarrow$ short spread
  \item Standard carry story: the near leg converges to spot
  \item \key{Result:} gross $-0.48\%$/yr, all-in net $-2.61\%$/yr;
        negative at zero cost. \key{Closed.}
\end{itemize}
\vspace{0.5ex}
{\scriptsize\color{slate}The short-contango side is the problem: raw mean
$+2.4$ bp/day, $t=2.46$; the long side is $\approx 0$.}
\end{column}
\begin{column}{0.45\textwidth}
\footnotesize
\textbf{Our main design: A-flip}
\begin{itemize}
  \item Reverse the sign: contango $\rightarrow$ long spread;
        backwardation $\rightarrow$ short spread
  \item Empirically, contango spreads strengthen as the near contract
        converges, so the assignment's short leg is on the wrong side
  \item Related: \textcite{koijen2018carry} document the standard commodity
        carry premium in A's direction. The flip is a data-driven reversal,
        not the standard carry trade.
\end{itemize}
\end{column}
\end{columns}
\vspace{0.5ex}
\centering

\toprule
 & Contango ($C_t<0$) & Backwardation ($C_t>0$) \\

\end{tabular}
\par}

\section{C: Spread mean reversion}

\begin{frame}{C \textperiodcentered\ Paper: mean reversion with dynamic hedge ratios}
\paperref{lubnau2015trading}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item State-space hedge ratio between near $A$ and far $B$:
  \begin{align*}
    p_{A,t} &= \beta_{1,t} + \beta_{2,t}\,p_{B,t} + v_t\\
    \beta_{j,t} &= \beta_{j,t-1} + w_{j,t}, \qquad j=1,2
  \end{align*}
  \item \where\ $\beta_{2,t}$ is the hedge ratio, filtered daily by the Kalman filter
  \item Hedged portfolio $p_{H,t} = p_{A,t} - \hat\beta_{2,t}\,p_{B,t}$
  \item Signal: $z_t = (p_{H,t} - \mathrm{MA}_{L,t})/\sigma_{L,t}$
  \item \key{Long $z_t<-2$, short $z_t>2$, exit at $z_t=0$}
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
\begin{itemize}
  \item Energy: WTI, natural gas, heating oil, gasoline
  \item Front month vs months 2--5; 1992--2013
  \item Holding 2--21 days; 0.1\% cost per trade
  \item Paper reports \key{Sharpe above 2} for most WTI and NG pairs
  \item Caveat: a change in $\hat\beta_{2,t}$ moves $p_{H,t}$ without P\&L
\end{itemize}
\vspace{1ex}
\begin{block}{Key idea}
{\scriptsize Trade the residual, not the price level.}
\end{block}
\end{column}
\end{columns}
\end{frame}
