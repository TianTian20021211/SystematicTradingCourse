\section{Setup}

\begin{frame}{Project 7: what we study}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item A futures curve prices the \key{same commodity} at different delivery months
  \item A calendar spread trades two points on that curve
  \[
    C_t = F(t,T_1) - F(t,T_2), \qquad T_1 < T_2
  \]
  \item Long the spread means long the near contract and short the far one,
        with P\&L $q\,M\,\Delta C_{t+h}$
  \item The flat price largely cancels, and what remains is the \key{shape of the curve},
        roll and convergence
  \item Research question: does term-structure information predict the spread
        change, \key{net of rolls and costs}?
\end{itemize}
\end{column}

\caption{Stylized futures curves. The arrow is the calendar spread.}

  \item The signal uses the $t$ settle and the order \key{fills at the $t+1$

\section{A: Spread convergence}

\begin{frame}{A \textperiodcentered\ Spread convergence: the idea}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item A calendar spread should \key{converge}: the near and far contracts
        cannot stay apart forever
  \item Spread: $C_t = F_{\text{near}} - F_{\text{far}}$, and convergence means
        $C_t \rightarrow 0$
  \item If $C_t < 0$ (contango), the spread is cheap: \key{long the spread}
  \item If $C_t > 0$ (backwardation), the spread is rich: \key{short the spread}
  \item Trade the main pair: long the near contract and short the far one,
        or the reverse
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{table}
\centering
\scriptsize
\setlength{\tabcolsep}{5pt}
\begin{tabular}{@{}l l@{}}
\toprule
State & Position \\
\midrule
Contango ($C_t<0$)      & long spread \\
Backwardation ($C_t>0$) & short spread \\
\bottomrule
\end{tabular}
\caption{Position by curve state.}
\end{table}
\vspace{0.5ex}
\begin{block}{Related paper}
{\scriptsize \textcite{kanamura2010profit} model spread trading as a
mean-reverting process with first-hitting times. In energy futures, profits
depend on spread volatility and the speed of mean reversion.}
\end{block}
\end{column}

\begin{frame}{B \textperiodcentered\ Curve relative value convergence: the idea}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize

\section{C: Spread mean reversion}

\begin{frame}{C \textperiodcentered\ Spread mean reversion: the idea}
\paperref{lubnau2015trading}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.56\textwidth}
\footnotesize
\begin{itemize}
  \item The trade is the \key{residual}, not the price level: hedge the near
        contract $A$ with the far contract $B$,
        $p_{H,t} = p_{A,t} - \beta_t\,p_{B,t}$
  \item Standardize the residual:
        $z_t = (p_{H,t} - \mathrm{MA}_{20,t})/\sigma_{20,t}$
  \item \key{Long $z_t<-2$, short $z_t>2$, exit at $z_t=0$}
  \item Three hedge ratios: \key{1:1 dollar} (headline),
        \key{60-day rolling OLS}, \key{Kalman DLM}
  \item 24 commodities, 1986--2026, main pair, weekly rebalance, $t+1$ fill,
        gross only
\end{itemize}
\end{column}
\begin{column}{0.41\textwidth}
\begin{table}

\section{D: Curve momentum}

\begin{frame}{D \textperiodcentered\ Curve momentum: the trade idea}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.57\textwidth}
\footnotesize
\begin{itemize}
  \item Within one commodity's curve, compare the two nearest tradable
        contracts (near = earlier expiry, far = later expiry). This is a
        \key{relative} trade, not a long or short commodity bet
  \item For each leg, compound its roll-adjusted daily rank return over the
        past \key{12 months}:
        $m_{i,t}=\prod_{s\in[t-12m,\,t]}(1+r_{i,s})-1$
  \item Signal: \key{$score_t=m_{near,t}-m_{far,t}$}
  \item If $score_t>0$, \key{long the near leg, short the far leg}.
        If $score_t<0$, reverse. Hold \key{one month}
  \item Month-end close signal, \key{$t+1$ close fill}. The target is held
        until the next month-end, monthly rebalance
  \item Economic story: relative curve momentum persists over the next month,
        and the contract that has outperformed stays stronger
\end{itemize}
\vspace{0.2ex}
{\scriptsize\color{slate}Reference: \textcite{paschke2020curve}.}

\section{E: Nelson--Siegel dynamics}

\begin{frame}{E \textperiodcentered\ Term-structure trading with Nelson--Siegel factors}
\paperref{bianchi2023exploiting}
\vspace{-0.8ex}
\begin{columns}[c, onlytextwidth]
\begin{column}{0.57\textwidth}
\footnotesize
\begin{itemize}
  \item \key{Data}: settlement prices of the four nearest eligible contracts,
        maturity $M$ in months to the last trading day
  \item \key{Model} from fixed income
        \parencite{nelson1987parsimonious, diebold2006forecasting}, OLS on each
        commodity-day, $\lambda$ chosen causally from a rolling 252-day fit
\end{itemize}
\vspace{-0.4ex}
{\scriptsize
\begin{equation}\label{eq:ns}
F_t(M)=\beta_{L,t}
+\beta_{S,t}\,\frac{1-e^{-\lambda M}}{\lambda M}
+\beta_{C,t}\Bigl(\frac{1-e^{-\lambda M}}{\lambda M}-e^{-\lambda M}\Bigr)
\end{equation}}
\vspace{-1.6ex}
\begin{itemize}
  \item \key{Three factors}
  \begin{itemize}\scriptsize
    \item Level $\beta_L$: the long-end price
    \item Slope $\beta_S=F_t(0)-\beta_L$: $>0$ backwardation (scarce inventory),
          $<0$ contango
    \item Curvature $\beta_C$: a hump or dip in the middle of the curve
  \end{itemize}
  \item \key{Trade, time series per commodity}: \hl{if $\Delta\beta_{S,t}>0$,
        long the 1st and short the 4th contract, else the reverse}. Fill at
        $t+1$, hold one day, $1/N$ budget, equal-notional legs
\end{itemize}
\end{column}
\begin{column}{0.41\textwidth}
\begin{figure}
\centering
\includegraphics[width=0.94\linewidth]{figures/e_curve_snapshots.pdf}
\caption{WTI crude oil curves on three dates. Dots are the four contracts.}
\end{figure}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{E \textperiodcentered\ Does Eq.~(\ref{eq:ns}) describe futures curves?}
\footnotesize
\begin{itemize}
  \item \textcite{nelson1987parsimonious} model the fixed-income forward curve,
        Eq.~(\ref{eq:nsfwd}), and Eq.~(\ref{eq:ns}) is its average over $[0,M]$.
        A futures price is not such an average, so \key{Eq.~(\ref{eq:ns}) need
        not hold}
        {\scriptsize
        \begin{equation}\label{eq:nsfwd}
        f(M)=\beta_L+\beta_S\,e^{-\lambda M}+\beta_C\,\lambda M\,e^{-\lambda M}
        \end{equation}}
        \vspace{-2.6ex}
  \item Only futures data can tell whether \key{each added factor} describes
        the term structure \key{significantly better}
  \item Nested fits $\beta_L$, then $\beta_L+\beta_S$, then Eq.~(\ref{eq:ns}),
        scored by the \key{out-of-sample $R^2$ of the held-out 2nd or 3rd
        contract}. Holding out the 1st or 4th is extrapolation, where $\beta_C$
        helps 10 commodities and hurts 10
\end{itemize}
\vspace{-1.4ex}
\begin{figure}
\centering
\includegraphics[width=0.85\linewidth]{figures/e_r2_gains.pdf}
\caption{Median daily gain in interior leave-one-contract-out $R^2$, 1986--2026,
21-day block bootstrap. Adding $\beta_S$ helps all 24 commodities, adding
$\beta_C$ helps 23.}
\end{figure}
\end{frame}

\section{Combining the alphas}

\begin{frame}{Combining the ideas: from strategies to alphas}
\footnotesize
\begin{itemize}
  \item Each idea makes economic sense, but each is \key{weak and works only in
        some regimes}: the A and B flips are gross only, C is eaten by costs,
        D is unstable, E works only at the paper's timing
  \item So we \key{distil every paper's strategy into continuous alphas} on the
        same main pair, combine them with a model, and trade the forecast
\end{itemize}
\vspace{-0.4ex}
{\scriptsize
With $r_t$ the main-pair spread return, $C_t$ its annualized carry, $c^{(k)}_t$
the carry of the $k$-th eligible adjacent spread, and $S_t$ the roll-chained
spread index:
\begin{align*}
\text{A:}\quad & a^{\text{carry}}_t=C_t, &
                 & a^{\text{carry,dev}}_t=C_t-\overline{C}_{t,252}\\
\text{B:}\quad & b^{\text{curve}}_t=c^{(\text{front})}_t-c^{(\text{back})}_t, &
                 & b^{\text{rel}}_t=C_t-\tfrac{1}{K}\textstyle\sum_k c^{(k)}_t\\
\text{C:}\quad & c^{z,L}_t=-\dfrac{S_t-\overline{S}_{t,L}}{\operatorname{sd}_{t,L}(S)},\ L\in\{20,60\}, &
                 & c^{\text{rev}}_t=-\textstyle\sum_{s=t-4}^{t}r_s\\
\text{D:}\quad & d^{\text{mom},L}_t=\textstyle\sum_{s=t-L+1}^{t}r_s,\ L\in\{21,63,252\} & & \\
\text{E:}\quad & e^{S}_t=\Delta\beta_{S,t}/\overline{F}_t, &
                 & e^{C}_t=\Delta\beta_{C,t}/\overline{F}_t
\end{align*}}
\vspace{-0.6ex}
{\scriptsize\color{slate}Each alpha uses settles up to $t$, is scaled per commodity by its
expanding root mean square and clipped at $\pm4$. Signs are kept, the model learns
the direction.}
\end{frame}

\begin{frame}{Is an alpha informative? Time-series IC with a day-block bootstrap}
\footnotesize
\begin{columns}[c, onlytextwidth]
\begin{column}{0.55\textwidth}
\textbf{Metric}: rank IC of alpha $a$ with the spread return over the next
$h$ days, per commodity $r$ over time, then averaged
{\scriptsize
\[
\mathrm{IC}_r(h)=\operatorname{corr}_t\bigl(\operatorname{rank}a_{r,t},\,
\operatorname{rank}R_{r,[t+1,\,t+1+h]}\bigr),\quad
\overline{\mathrm{IC}}(h)=\tfrac{1}{N}\textstyle\sum_r\mathrm{IC}_r(h)
\]}
\vspace{-1.4ex}
\begin{block}{Joint day-block bootstrap, train 1986--2005}
\scriptsize
\textbf{for} $b=1,\dots,B$ ($B=1000$ replications)\textbf{:}\\
\quad draw $\lceil T/L\rceil$ blocks of $L$ consecutive dates, $T$ dates in total,\\
\quad the same dates for every commodity and every alpha\\
\quad $\mathrm{IC}^{*}_{r}\leftarrow$ rank correlation on the resampled dates\\
\quad $\overline{\mathrm{IC}}^{*}\leftarrow$ mean of $\mathrm{IC}^{*}_{r}$ over commodities\\
s.e.\ $=\operatorname{sd}(\overline{\mathrm{IC}}^{*})$,\quad
$p_r=2\min\{\Pr(\mathrm{IC}^{*}_r\le0),\,\Pr(\mathrm{IC}^{*}_r\ge0)\}$
\end{block}
\vspace{-0.4ex}
\begin{itemize}
  \item \key{Keep} an alpha if $p_r<0.05$ with the mean's sign on half of the
        commodities, else if the CI of $\overline{\mathrm{IC}}$ excludes 0 and
        two thirds share its sign
\end{itemize}
\end{column}

\begin{frame}{Appendix: cost model and definitions}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.48\textwidth}
\footnotesize
\begin{itemize}
  \item \key{Half-spread}: per-contract quote, paid on each leg
  \item \key{Fee}: exchange fee per contract
  \item \key{Impact}: square-root of order size / ADV
  \item All-in net = gross $-$ half-spread $-$ fee $-$ impact
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\footnotesize
\begin{itemize}
  \item \key{Eligibility}: no delivery month, exit at least 5 trading days
        before it, and at least 5 trading days left to the last trading day
  \item \key{Main pair}: two nearest eligible contracts
  \item \key{Carry}: annualized near--far price difference
  \item \key{Test}: 2016--2026, read once and never used for tuning
\end{itemize}
\end{column}
\end{columns}
