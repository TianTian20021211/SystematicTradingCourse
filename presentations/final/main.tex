\section{Setup}

\begin{frame}{Project 7: what we study}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item A futures curve prices the \key{same commodity} at different delivery months
  \item A calendar spread trades two points on that curve
  \[
    C_t = F(t,T_1) - F(t,T_2), \qquad T_1 < T_2
  \]
  \item Long the spread means long the near contract and short the far one,
        with P\&L $q\,M\,\Delta C_{t+h}$
  \item The flat price largely cancels, and what remains is the \key{shape of the curve},
        roll and convergence
  \item Research question: does term-structure information predict the spread
        change, \key{net of rolls and costs}?
\end{itemize}
\end{column}

\centering
\includegraphics[width=\linewidth]{figures/stylized_curves.pdf}
\caption{Stylized futures curves. The arrow is the calendar spread.}
\end{figure}
\end{column}
\end{columns}
\end{frame}

\section{A: Carry timing}

\begin{frame}{A \textperiodcentered\ Carry: the assignment direction, then the flip}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.52\textwidth}
\footnotesize
\textbf{Original A} (assignment Example 1)
\begin{itemize}
  \item Backwardation ($C_t>0$) $\rightarrow$ long spread,
        contango ($C_t<0$) $\rightarrow$ short spread
  \item Standard carry story: the near leg converges to spot
  \item \key{Result:} gross $-0.48\%$/yr, all-in net $-2.61\%$/yr,
        negative at zero cost. \key{Closed.}
\end{itemize}
\vspace{0.5ex}
{\scriptsize\color{slate}The short-contango side is the problem: raw mean
$+2.4$ bp/day, $t=2.46$, while the long side is $\approx 0$.}
\end{column}
\begin{column}{0.45\textwidth}
\footnotesize
\textbf{Our main design: A-flip}
\begin{itemize}
  \item Reverse the sign: contango $\rightarrow$ long spread,
        backwardation $\rightarrow$ short spread
  \item Empirically, contango spreads strengthen as the near contract
        converges, so the assignment's short leg is on the wrong side
  \item Related: \textcite{koijen2018carry} document the standard commodity
        carry premium in A's direction. The flip is a data-driven reversal,
        not the standard carry trade.
\end{itemize}
\end{column}
\end{columns}

\textbf{Original B} (assignment Example 2)
\begin{itemize}
  \item At each date and root, rank annualized carry of

\section{C: Spread mean reversion}

\begin{frame}{C \textperiodcentered\ Paper: mean reversion with dynamic hedge ratios}
\paperref{lubnau2015trading}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{itemize}
  \item State-space hedge ratio between near $A$ and far $B$:
  \begin{align*}
    p_{A,t} &= \beta_{1,t} + \beta_{2,t}\,p_{B,t} + v_t\\
    \beta_{j,t} &= \beta_{j,t-1} + w_{j,t}, \qquad j=1,2
  \end{align*}
  \item \where\ $\beta_{2,t}$ is the hedge ratio, filtered daily by the Kalman filter
  \item Hedged portfolio $p_{H,t} = p_{A,t} - \hat\beta_{2,t}\,p_{B,t}$
  \item Signal: $z_t = (p_{H,t} - \mathrm{MA}_{L,t})/\sigma_{L,t}$
  \item \key{Long $z_t<-2$, short $z_t>2$, exit at $z_t=0$}
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\footnotesize
\begin{itemize}
  \item Energy: WTI, natural gas, heating oil, gasoline
  \item Front month vs months 2--5, 1992--2013
  \item Holding 2--21 days, 0.1\% cost per trade
  \item Paper reports \key{Sharpe above 2} for most WTI and NG pairs
  \item Caveat: a change in $\hat\beta_{2,t}$ moves $p_{H,t}$ without P\&L
\end{itemize}
\vspace{1ex}
\begin{block}{Key idea}
{\scriptsize Trade the residual, not the price level.}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{E: Nelson--Siegel dynamics}

\begin{frame}{E \textperiodcentered\ Term-structure trading with Nelson--Siegel factors}
\paperref{bianchi2023exploiting}
\vspace{-0.8ex}
\begin{columns}[T, onlytextwidth]
\begin{column}{0.57\textwidth}
\footnotesize
\begin{itemize}
  \item \key{Data}: settlement prices of the four nearest eligible contracts,
        maturity $M$ in months to the last trading day
  \item \key{Model} from fixed income
        \parencite{nelson1987parsimonious, diebold2006forecasting}, OLS on each
        commodity-day, $\lambda$ chosen causally from a rolling 252-day fit
\end{itemize}
\vspace{-0.4ex}
{\scriptsize
\begin{equation}\label{eq:ns}
F_t(M)=\beta_{L,t}
+\beta_{S,t}\,\frac{1-e^{-\lambda M}}{\lambda M}
+\beta_{C,t}\Bigl(\frac{1-e^{-\lambda M}}{\lambda M}-e^{-\lambda M}\Bigr)
\end{equation}}
\vspace{-1.6ex}
\begin{itemize}
  \item \key{Three factors}
  \begin{itemize}\scriptsize
    \item Level $\beta_L$: the long-end price
    \item Slope $\beta_S=F_t(0)-\beta_L$: $>0$ backwardation (scarce inventory),
          $<0$ contango
    \item Curvature $\beta_C$: a hump or dip in the middle of the curve
  \end{itemize}
  \item \key{Trade, time series per commodity}: if $\Delta\beta_{S,t}>0$, long
        the 1st and short the 4th contract, else the reverse. Fill at $t+1$,
        hold one day, $1/N$ budget, equal-notional legs
\end{itemize}
\end{column}
\begin{column}{0.41\textwidth}
\begin{figure}
\centering
\includegraphics[width=0.94\linewidth]{figures/e_curve_snapshots.pdf}
\caption{WTI crude oil curves on three dates. Dots are the four contracts.}
\end{figure}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{E \textperiodcentered\ Does Eq.~(\ref{eq:ns}) describe futures curves?}
\footnotesize
\begin{itemize}
  \item \textcite{nelson1987parsimonious} model the fixed-income forward curve,
        Eq.~(\ref{eq:nsfwd}), and Eq.~(\ref{eq:ns}) is its average over $[0,M]$.
        A futures price is not such an average, so Eq.~(\ref{eq:ns}) need not hold
        {\scriptsize
        \begin{equation}\label{eq:nsfwd}
        f(M)=\beta_L+\beta_S\,e^{-\lambda M}+\beta_C\,\lambda M\,e^{-\lambda M}
        \end{equation}}
        \vspace{-2.6ex}
  \item Only futures data can tell whether each added factor describes the term
        structure significantly better
  \item Nested fits $\beta_L$, then $\beta_L+\beta_S$, then Eq.~(\ref{eq:ns}),
        scored by the out-of-sample $R^2$ of the held-out 2nd or 3rd contract.
        Holding out the 1st or 4th is extrapolation, where $\beta_C$ helps 10
        commodities and hurts 10
\end{itemize}
\vspace{-1.4ex}
\begin{figure}
\centering
\includegraphics[width=0.85\linewidth]{figures/e_r2_gains.pdf}
\caption{Median daily gain in interior leave-one-contract-out $R^2$, 1986--2026,
21-day block bootstrap. Adding $\beta_S$ helps all 24 commodities, adding
$\beta_C$ helps 23.}
\end{figure}
\end{frame}

\begin{frame}{E \textperiodcentered\ Replicating the paper's backtest, then making it tradable}
\footnotesize
\begin{columns}[T, onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{From today's curve to a position}
\begin{itemize}
  \item Fit Eq.~(\ref{eq:ns}) at the $t$ settle. If $\Delta\beta_{S,t}>0$, long
        the 1st and short the 4th contract, else the reverse

\end{column}
